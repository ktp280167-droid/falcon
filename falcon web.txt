    <script>
        const ageScreen = document.getElementById('ageScreen');
        const landingScreen = document.getElementById('landingScreen');
        const websiteWrapper = document.getElementById('websiteWrapper');
        const modal = document.getElementById('loginModal');
        const mainContent = document.getElementById('mainWebsiteContent');
        const adminDashboard = document.getElementById('adminDashboard');
        const ageError = document.getElementById('ageError');

        function verifyAge(isAbove18) {
            if (isAbove18) {
                ageScreen.style.display = 'none';
                landingScreen.style.display = 'flex';
            } else {
                ageError.style.display = 'block';
            }
        }

        function openAdminModal() {
            modal.style.display = 'flex';
        }

        function closeModal() {
            modal.style.display = 'none';
            document.getElementById('errorMsg').style.display = 'none';
            document.getElementById('loginForm').reset();
        }

        function loginAsGuest() {
            landingScreen.style.display = 'none';
            websiteWrapper.style.display = 'block';
            mainContent.style.display = 'block';
            adminDashboard.style.display = 'none';
        }

        function handleLogin(event) {
            event.preventDefault();
            const user = document.getElementById('username').value;
            const pass = document.getElementById('password').value;
            const errorMsg = document.getElementById('errorMsg');

            if (user === '1212' && pass === '1212') {
                closeModal();
                landingScreen.style.display = 'none';
                websiteWrapper.style.display = 'block';
                mainContent.style.display = 'none';
                adminDashboard.style.display = 'block';
            } else {
                errorMsg.style.display = 'block';
            }
        }

        function returnToLanding() {
            websiteWrapper.style.display = 'none';
            landingScreen.style.display = 'flex';
            document.getElementById('loginForm').reset();
        }

        function showMainSite() {
            adminDashboard.style.display = 'none';
            mainContent.style.display = 'block';
        }

        window.onclick = function(event) {
            if (event.target == modal) {
                closeModal();
            }
        }
    </script>
